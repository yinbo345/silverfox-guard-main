# SilverFoxGuard 勒索回滚引擎配置（可热更新）
#
# ===========================================================================
#  为什么单独出一个配置文件
# ===========================================================================
#  卡巴斯基的 System Watcher 官方文档明确写：受监控的事件类型、回滚策略、
#  处置动作（Action on detection / Action if can be rolled back）全部可配。
#  本文件对齐这一做法：调参只需改这个文本文件，不必重新编译/签名/安装。
#
#  加载时机：主防服务启动时读 data\rollback_rules.txt。
#           （候选路径：<安装目录>\data\ 、<安装目录>\ 、C:\ProgramData\SilverFoxGuard\）
#  文件缺失或单条非法 → 该项回退内置默认值，不阻断启动。
# ===========================================================================
#
#  格式：key=value，# 之后为注释，空行忽略
# ===========================================================================
#
#  ---------------------------------------------------------------------------
#  【快照缓存】
#  ---------------------------------------------------------------------------
#  总容量上限（MB）。超出后按 LRU 淘汰最旧的快照。
#
#  ⚠️ 2026-09-18 由 512 下调到 160（维护者实测反馈"快照占用太大"）。
#     配合下方「风险分层建快照」后，正常使用下的实际占用远低于此上限；
#     160MB 是"极端场景兜底"，而不是日常预期值。
max_cache_mb = 160

#  单文件快照体积上限（MB）。超过此值的文件不备份
#  ——避免一个虚拟机镜像把整个缓存吃光。
#
#  ⚠️ 2026-09-18 由 32 下调到 4。依据：用户真实文档 99% 在 4MB 以内，
#     而 4MB 以上多为视频/镜像/虚拟机盘——那些即便被加密，用户也大概率
#     不会指望"原样回滚"。收紧后单文件占用上限降到原来的 1/8。
max_file_mb = 4

#  ---------------------------------------------------------------------------
#  【风险分层建快照】（2026-09-18 新增，快照瘦身的核心）
#  ---------------------------------------------------------------------------
#  为什么需要：旧策略是"任何改写都建快照"，于是普通软件每次保存文档都存一份
#  （Word 自动保存、IDE 频繁落盘、浏览器缓存写入……），占用迅速堆积，还会把
#  更早的、真正有价值的快照挤掉。
#
#  开启后只在下列任一条件成立时才建快照：
#    ① 文件被改成**陌生后缀**（勒索加密后必然改名，正常软件保存不改扩展名）；
#    ② 该文件所在目录在 burst_window_sec 秒内已被改写达 burst_file_trigger 次
#       （这片区域正在被批量动，风险升高）；
#    ③ 文件属于**高价值文档**（office/pdf/图片/音视频/代码），且是本轮首次改写。
#
#  关闭（设 0）则回到旧行为：任何改写都建快照，占用量大。
risk_tiered_snapshot = 1

#  目录级"热度"观察窗口（秒）与触发次数。
#  依据：正常软件极少在 30 秒内对同一目录做 5 次以上的**不同文件**改写；
#        一旦达到，说明发生了批量操作（无论善恶意），此后该目录的改写全部建快照。
burst_window_sec = 30
burst_file_trigger = 5

#  是否只对高价值文档做"首次改写即存"（判据③）。
#  设 0 则连普通类型的首次改写也不存，占用更省但保护面更窄。
high_value_only = 1

#  ---------------------------------------------------------------------------
#  【勒索判定：滑动窗口】
#  ---------------------------------------------------------------------------
#  为什么要"窗口 + 多信号"而不是简单计数：
#  正常软件（编译器、解压工具、批量重命名）也会在短时间内改大量文件，
#  只看数量必然误报。业界共识是必须联合"高熵随机后缀改名"与"写勒索说明"
#  等旁证才能定性。窗口越短越灵敏但越容易误判，10 秒是折中值。
window_seconds = 10

#  窗口内被改写的文件数达到该值 → 触发"批量改写"信号。
#
#  ⚠️ 2026-09-19 由 25 上调到 40。原因：维护者实测反馈"一天一次误判"，
#     取证发现 Edge 在更新扩展/写缓存时会短时间改动 30~36 个文件，
#     恰好越过 25 这条线。真勒索的量级是**成百上千**（16:37 那次 2000 个），
#     40 这个值在"避开浏览器日常"与"仍能拦真勒索"之间留了足够距离。
files_threshold = 40

#  窗口内"改成高熵随机后缀"的次数达到该值 → 触发"改名"信号。
#
#  ⚠️ 2026-09-19 由 10 上调到 15。这是**判定成立的必要条件**（见下方说明），
#     所以它必须足够高，不能靠"几个文件的日志轮转"就凑满。
rename_threshold = 15

#  窗口内出现勒索说明文件（README / HOW TO DECRYPT / 解密说明）的次数。
#  ⚠️ 2026-09-19：note 信号**不再能单独配合少量改动定性**，
#     必须同时满足 rename_threshold，故此项仅作门槛值保留。
note_threshold = 2

#  ---------------------------------------------------------------------------
#  【2026-09-19 重大修正：判定必须基于"改名的物理结果"】
#  ---------------------------------------------------------------------------
#  事故回顾：维护者反馈"每天几乎都有一次 Edge 扩展被清空"。
#  根因是判定过松导致的**误覆盖**（不是删除）：回滚用 CREATE_ALWAYS 把旧快照
#  盖回新文件，Edge 的 Preferences（扩展注册表就在里面）被换回旧版本，
#  extensions.settings 退回旧状态 → 扩展"凭空消失"（文件系统无删除记录）。
#
#  旧判据的两条通道都过松：
#    · noteSignal && (mod+ren) >= 3
#        → 一个含 "readme" 的文件 + 任意 3 个文件被改 = 成立。
#          「装软件解压」「编译」「浏览器写缓存」全都满足。
#    · burst && renameSignal
#        → 依赖当时过宽的 LooksLikeRansomExt（压缩后缀未排除）与
#          LooksLikeRansomNote（子串匹配 "readme"）→ Edge 的
#          edge_BITS_* / scoped_dir* 随机名文件反复命中。
#
#  修正后的判定形状：
#    A. 批量改写（files_threshold） + 批量陌生后缀改名（rename_threshold）
#       —— 这两条**必须同时成立**，是可独立定性的完整勒索形态。
#    B. 勒索说明文件 + 批量陌生后缀改名
#       —— 说明文件只能作为加强旁证，不再可单独配合少量改动定性。
#
#  核心原则：**必须有"文件被改成陌生后缀"这一条不可伪装的硬证据**。
#  加密必然改名，正常软件保存不改扩展名 —— 这是唯一可靠的单一判据。
#
#  同时新增了浏览器数据目录整体排除（AppData\Local\Microsoft\Edge 等），
#  双保险确保浏览器配置/扩展状态不会被回滚触碰。

#  ---------------------------------------------------------------------------
#  【密钥截获与留存】（2026-09-19 新增）
#  ---------------------------------------------------------------------------
#  攻击链条：生成密钥 → 密钥落盘 → 加密全部文件 → **删除密钥** → 索赎金。
#  第 ④ 步是攻击者自保：密钥一删，抓到样本也解不开。
#  对我们而言，**第 ② 步是唯一窗口** —— 密钥此刻正以明文躺在磁盘上。
#
#  留存位置：C:\ProgramData\SilverFoxGuard\rollback_cache\keys\
#    · key_NNNN.bin      密钥副本（只读属性，抬高勒索删除门槛）
#    · manifest.ndjson   索引（追加写，服务重启后据此重建内存索引）
#
#  ⚠️ 本模块**只做留存，不做解密**。解密需要判断密钥算法/模式/IV，
#     猜错会把文件彻底弄坏。留存保住了可能性，解密交由人工取证。
key_hunt_enabled = 1

#  密钥体积区间（字节）。小于下限排除空文件/占位，大于上限不是密钥
#  （RSA-4096 私钥 PEM 约 3.2KB，64KB 已非常宽松）。
key_min_bytes = 16

#  密钥最小熵值（bit/byte）。随机密钥应 >= 7.5。
key_entropy_min = 7.5

#  密钥落盘后多久内出现**同目录**批量改写算作"时序关联成立"。
#  ⚠️ 2026-09-19：关联范围收紧为"仅同目录"（旧版把 Temp/盘根的候选
#     视为与任意目录相关，导致 Edge 的 edge_BITS_* 天天触发提前定性）。
key_window_sec = 30

#  最多保留多少份密钥副本。
key_max_kept = 32

#  密钥副本是否加只读属性。勒索者删密钥用普通 DeleteFile，
#  对只读文件会失败；多数勒索实现不做"先清只读再删"。
key_readonly_guard = 1

#  ---------------------------------------------------------------------------
#  【文件落地前置捕获】（2026-09-19 新增）
#  ---------------------------------------------------------------------------
#  银狐的攻击链条是「诱导下载 → 载荷落盘 → 自启执行」。现有事件源
#  （WMI 进程创建 + 注册表 Run）都在第三步之后 —— 载荷已经在跑了。
#  把判定点前移到「落盘那一刻」，可在攻击者还没执行时就已定性。
#
#  判据：① 落地位置（Temp/下载/盘根/桌面）
#        + （② PE 头/脚本头 或 ③ 随机命名）→ 记入候选。
land_hunt_enabled = 1
land_max_mb = 64

#  ---------------------------------------------------------------------------
#  【回滚前备份】（2026-09-21 新增）
#  ---------------------------------------------------------------------------
#  【为什么需要：回滚是「误覆盖」而不是「误删」】
#  回滚用 CREATE_ALWAYS 把旧快照**无条件截断重写**到目标文件上。文件系统
#  层面看不到任何删除动作 —— 磁盘上那一份更好的内容是在同一个句柄里被
#  新内容盖掉的，回收站、卷影副本、文件历史统统帮不上忙。
#
#  实测教训（Edge 扩展集体消失事件）：引擎误判后把 Preferences 换回旧版，
#  扩展记录随之蒸发。用户想找回被覆盖的那一份，但它在覆盖瞬间就没了。
#
#  开启后，每一次覆盖写之前都会把即将被抹掉的内容存一份到
#      <rollback_cache>\prebackup#  事后可用 GUI 的「回滚前备份」列表逐条还原。
#
#  ⚠️ 与 undo 目录的区别（不是重复功能）：
#     undo     只在**勒索处置回滚**时留存，10 分钟窗口，随回滚记录清空，
#              目的是让用户撤销本次处置。
#     prebackup 在**每一次覆盖写之前**留存，独立生命周期（按小时轮转），
#              目的是无论谁触发的覆盖，都能把覆盖前那一版捞回来。
#     关键是 undo 靠上层自觉调用，而 prebackup 挂在真正落笔的那一步 ——
#     将来新增的回滚路径不会因为忘了加备份而失去兜底。
rollback_pre_backup = 1

#  备份失败时是否**放弃本次覆盖写**。
#    0（默认）= 备份失败只记日志，回滚照常执行（回滚比备份重要）。
#    1（严格）= 备份失败即中止该文件的覆盖，宁可这文件不回滚，
#               也保证磁盘上没有任何一份内容被无声抹掉。
#  ⚠️ 严格模式会让备份失败变成救援失败，请确认能接受再开。
rollback_pre_backup_strict = 0

#  备份区容量上限（MB）。与 160MB 快照缓存**独立计算**，互不挤占。
#  超出后从最旧的开始淘汰。
rollback_pre_backup_mb = 64

#  备份保留期（小时）。超期由巡逻线程自动清理（每 30 秒一轮）。
#  设为 0 表示不按时间淘汰，只受上面的容量上限约束。
rollback_pre_backup_hours = 24

#  单文件备份体积上限（MB）。超过则不备份 —— 避免一个 .iso 把 64MB 配额
#  一次吃光。注意：超限时**严格模式也不阻断**回滚，因为大文件中招也是
#  真实场景，不该因为备份不了就放弃救援。
rollback_pre_backup_file_mb = 16

#  ---------------------------------------------------------------------------
#  【熵值阈值】
#  ---------------------------------------------------------------------------
#  Shannon 熵，单位 bit/byte，范围 0..8。
#  加密数据、压缩数据的典型熵值在 7.6 以上；普通文档约 3~6。
#  ⚠️ 单独看熵值**不足以**判定勒索（zip/7z/jpg/mp4 本身就接近 8.0），
#     本引擎只在配合"批量改动 + 陌生后缀改名"时才采用该信号。
entropy_threshold = 7.6

#  ---------------------------------------------------------------------------
#  【处置动作】（对齐卡巴的两档配置）
#  ---------------------------------------------------------------------------
#  Action on detection：检测到勒索后是否终止肇事进程。
#  1 = 终止（推荐，卡巴的 Action on detection 默认即"终止应用"）
#  0 = 不终止（仅回滚文件，适用于调试）
terminate_on_detect = 1

#  Action if can be rolled back：是否启用回滚。
#  本引擎的存在意义即回滚，故设为 0 时整个引擎不启动监控线程（省资源）。
enabled = 1

#  ---------------------------------------------------------------------------
#  【已知的局限性（诚实标注，不做虚假承诺）】
#  ---------------------------------------------------------------------------
#  1. 本引擎是**纯用户态**实现，不像卡巴那样有内核微过滤驱动在文件系统层拦写。
#     因此存在"被加密内容抢先写入"的竞态窗口（实测通知延迟约 1ms 量级）。
#     缓解手段是周期性基线巡逻：对尚无快照的受保护文件预先建立基线，
#     即使某次写前快照来晚了，也还能从更早的基线恢复。
#  2. 卡巴官方文档也承认回滚存在边界："有些情况下应用的活动无法回滚，
#     例如应用被 Intrusion Prevention 组件检测到时。"
#     本引擎对应的边界是：**没有快照的文件一律不可恢复**，会在报告中
#     如实列出清单，绝不尝试"猜测还原"（那是数据破坏，不是恢复）。
#  3. 快照缓存有容量上限，超出即淘汰最旧。淘汰掉的文件在后续勒索中
#     将不可恢复——这是"占盘"与"保护窗口"之间的必然取舍。
# ===========================================================================
